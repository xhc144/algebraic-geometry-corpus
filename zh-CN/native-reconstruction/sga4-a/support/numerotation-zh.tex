\newcounter{pageoriginal}
\newcommand{\pageoriginale}{\refstepcounter{pageoriginal}\marginpar{\footnotesize\textbf{\thepageoriginal}}\xspace}

\usepackage{ifthen}
\newboolean{original}
\def\sisi{\ifthenelse{\boolean{original}}}
\setboolean{original}{false}

\newcounter{cnote}
\def\nde#1{\ifthenelse{\boolean{original}}{}{\stepcounter{cnote}\begingroup\def\thefootnote{\arabic{cnote}}\protect\footnote[\value{cnote}]{编者注: #1}\endgroup}}
\def\ndemark{\ifthenelse{\boolean{original}}{}{\stepcounter{cnote}\begingroup\def\thefootnote{\arabic{cnote}}\protect\footnotemark[\value{cnote}]\endgroup}}
\def\ndetext#1{\ifthenelse{\boolean{original}}{}{\begingroup\def\thefootnote{\arabic{cnote}}\protect\footnotetext[\value{cnote}]{编者注: #1}\endgroup}}
